
    function renderStockList(stockData) {
        const tbody = document.getElementById('stock-list-body');
        const badge = document.querySelector('.count-badge');
        if (!tbody) return;
        
        tbody.innerHTML = '';
        if (badge) badge.textContent = stockData.length + '点';

        if (stockData.length === 0) {
            const msg = currentMasters['T_在庫集計'] ? '表示できる在庫はありません' : '在庫データを取得できませんでした。';
            tbody.innerHTML = `<tr><td colspan="5" style="text-align:center; padding: 40px; color: var(--text-secondary);">${msg}</td></tr>`;
            return;
        }

        stockData.forEach(row => {
            const itemName = row['品名'];
            const rawQty = parseFloat(row['現在庫数']);
            const stockQty = isNaN(rawQty) ? 0 : rawQty;
            
            const rawThreshold = parseFloat(row['閾値']);
            const threshold = isNaN(rawThreshold) ? 10 : rawThreshold;
            const useFlag = parseInt(row['使用FLG']) !== 0;
            
            const tr = document.createElement('tr');
            if (!useFlag) tr.classList.add('inactive-row');

            const itemInMaster = (currentMasters['M_商品'] || []).find(m => m['品名'] === itemName);
            const imageUrl = itemInMaster ? itemInMaster['画像URL'] : null;

            // 状態判定
            let isDanger = stockQty < threshold;

            tr.innerHTML = `
                <td class="product-name" style="display: flex; align-items: center;">
                    <div class="product-thumb-container" onclick="showImageModal('${imageUrl}')">
                        ${imageUrl ? `<img src="${imageUrl}" loading="lazy">` : `<ion-icon name="image-outline"></ion-icon>`}
                    </div>
                    ${itemName}
                </td>
                <td class="stock-val ${!isDanger ? 'safe' : ''}">${stockQty}</td>
                <td><input type="number" class="table-input" value="${threshold}" data-original="${threshold}"></td>
                <td class="status-cell">
                    <ion-icon name="${isDanger ? 'alert-circle' : 'checkmark-circle'}" 
                               class="status-icon ${isDanger ? 'danger' : 'success'}"
                               title="${isDanger ? '在庫不足' : '在庫あり'}"></ion-icon>
                </td>
                <td class="action-col">
                    <button class="camera-btn" onclick="triggerPhotoUpload('${itemName}')" title="写真を登録/変更">
                        <ion-icon name="camera-outline"></ion-icon>
                    </button>
                    <button class="update-mini-btn no-text btn-save-threshold" title="数値を保存">
                        <ion-icon name="save-outline"></ion-icon>
                    </button>
                    <button class="update-mini-btn no-text secondary-action btn-archive-item" title="${useFlag ? '廃盤にする' : '復活させる'}">
                        <ion-icon name="${useFlag ? 'archive-outline' : 'refresh-outline'}"></ion-icon>
                    </button>
                </td>
            `;

            // --- イベントリスナー ---
            const input = tr.querySelector('.table-input');
            const statusCell = tr.querySelector('.status-cell');
            const stockValCell = tr.querySelector('.stock-val');
            const saveBtn = tr.querySelector('.btn-save-threshold');
            const archiveBtn = tr.querySelector('.btn-archive-item');

            // 1. 閾値入力時のリアルタイムアイコン更新
            input.addEventListener('input', (e) => {
                const newThreshold = parseFloat(e.target.value);
                const original = parseFloat(e.target.getAttribute('data-original'));
                
                // アイコン更新
                const danger = stockQty < newThreshold;
                statusCell.innerHTML = `
                    <ion-icon name="${danger ? 'alert-circle' : 'checkmark-circle'}" 
                               class="status-icon ${danger ? 'danger' : 'success'}"
                               title="${danger ? '在庫不足' : '在庫あり'}"></ion-icon>
                `;
                stockValCell.classList.toggle('safe', !danger);

                // 未保存状態の視覚化
                const isDirty = (newThreshold !== original && !isNaN(newThreshold));
                input.classList.toggle('is-dirty', isDirty);
                saveBtn.classList.toggle('is-dirty', isDirty);
            });

            // 2. 閾値の保存
            saveBtn.addEventListener('click', async () => {
                const newThreshold = parseFloat(input.value);
                if (isNaN(newThreshold)) return;

                const originalHTML = saveBtn.innerHTML;
                saveBtn.innerHTML = '<ion-icon name="sync-outline" class="spinning"></ion-icon>';
                saveBtn.disabled = true;

                try {
                    const res = await fetchAPI('updateStockThreshold', { itemName: itemName, threshold: newThreshold });
                    if (res.status === 'success') {
                        input.setAttribute('data-original', newThreshold);
                        input.classList.remove('is-dirty');
                        saveBtn.classList.remove('is-dirty');
                        saveBtn.innerHTML = '<ion-icon name="checkmark-outline"></ion-icon>';
                        saveBtn.style.backgroundColor = 'var(--accent-green)';
                        
                        const masterIdx = currentMasters['T_在庫集計'].findIndex(m => m['品名'] === itemName);
                        if (masterIdx !== -1) currentMasters['T_在庫集計'][masterIdx]['閾値'] = newThreshold;

                        setTimeout(() => {
                            saveBtn.innerHTML = originalHTML;
                            saveBtn.style.backgroundColor = '';
                            saveBtn.disabled = false;
                        }, 1500);
                    } else {
                        throw new Error(res.message);
                    }
                } catch (e) {
                    alert('保存失敗: ' + e.message);
                    saveBtn.innerHTML = originalHTML;
                    saveBtn.disabled = false;
                }
            });

            // 3. アーカイブ / 復元
            archiveBtn.addEventListener('click', async () => {
                const actionText = useFlag ? '廃盤（一覧から非表示）' : '有効化';
                if (!confirm(`「${itemName}」を${actionText}にしますか？`)) return;

                const originalHTML = archiveBtn.innerHTML;
                archiveBtn.innerHTML = '<ion-icon name="sync-outline" class="spinning"></ion-icon>';
                archiveBtn.disabled = true;

                try {
                    const newStatus = useFlag ? 0 : 1;
                    const res = await fetchAPI('updateStockItemStatus', { itemName: itemName, status: newStatus });
                    if (res.status === 'success') {
                        const masterIdx = currentMasters['T_在庫集計'].findIndex(m => m['品名'] === itemName);
                        if (masterIdx !== -1) currentMasters['T_在庫集計'][masterIdx]['使用FLG'] = newStatus;
                        
                        const stockSearchInput = document.getElementById('stock-search-input');
                        stockSearchInput.dispatchEvent(new Event('input')); 
                    } else {
                        throw new Error(res.message);
                    }
                } catch (e) {
                    alert('更新失敗: ' + e.message);
                    archiveBtn.innerHTML = originalHTML;
                    archiveBtn.disabled = false;
                }
            });

            tbody.appendChild(tr);
        });
    }
